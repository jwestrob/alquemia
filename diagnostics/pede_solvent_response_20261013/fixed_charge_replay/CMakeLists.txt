cmake_minimum_required(VERSION 3.17)
project(fixed_charge_replay LANGUAGES C Fortran)
set(WITH_JSON OFF CACHE BOOL "No optional JSON dependency" FORCE)
set(WITH_TBLITE OFF CACHE BOOL "Native xTB component path only" FORCE)
set(WITH_CPCMX OFF CACHE BOOL "Native ALPB component path only" FORCE)
set(WITH_OpenMP OFF CACHE BOOL "One numerical thread per replay" FORCE)
set(WITH_OBJECT TRUE CACHE BOOL "Original library layout" FORCE)
set(FETCHCONTENT_FULLY_DISCONNECTED ON CACHE BOOL "No unpinned downloads" FORCE)
add_subdirectory("${SOURCE_ROOT}/test-drive" test-drive EXCLUDE_FROM_ALL)
add_library(test-drive::test-drive ALIAS test-drive)
add_subdirectory("${SOURCE_ROOT}/mctc-lib" mctc-lib EXCLUDE_FROM_ALL)
add_library(mctc-lib::mctc-lib ALIAS mctc-lib)
add_subdirectory("${SOURCE_ROOT}/xtb" xtb EXCLUDE_FROM_ALL)
add_executable(replay REPLAY.f90)
target_link_libraries(replay PRIVATE lib-xtb-static)
target_compile_options(replay PRIVATE -ffree-line-length-none)
set_target_properties(replay PROPERTIES BUILD_RPATH "${BLAS_RUNTIME_DIR}")
